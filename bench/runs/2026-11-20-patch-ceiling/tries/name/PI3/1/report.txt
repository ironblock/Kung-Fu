export function visitsOf(log: Visit[], boat: string, latest?: number): Visit[] {
  const all = log.filter(v => v.boat === boat); // keep the existing matching logic
  if (latest === undefined) return all;
  return latest <= 0 ? [] : all.slice(-latest);
}

- **Why `slice(-latest)`:** Visits only enter the log through the module's recording function and are never moved, so log order is the order they were recorded. The last `latest` matches are therefore the most recently recorded, still in log order. Day and start time are not used.
